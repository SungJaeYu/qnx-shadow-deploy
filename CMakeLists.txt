cmake_minimum_required(VERSION 3.20)
project(qnx_shadow_deploy VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SHADOW_SANITIZERS "Enable address and undefined behavior sanitizers on the host" OFF)
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(SHADOW_SANITIZERS AND NOT CMAKE_SYSTEM_NAME STREQUAL "QNX")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "QNX")
  add_compile_definitions(_QNX_SOURCE)
endif()
add_library(shadow_core src/policy.cpp)
target_include_directories(shadow_core PUBLIC include)
add_library(shadow_transport src/process.cpp src/io.cpp)
target_include_directories(shadow_transport PUBLIC include)
target_link_libraries(shadow_transport PUBLIC Threads::Threads)
add_library(shadow_staging src/staging.cpp)
target_link_libraries(shadow_staging PUBLIC shadow_core shadow_transport)
add_executable(shadow-worker src/worker.cpp src/io.cpp)
target_include_directories(shadow-worker PRIVATE include)
target_link_libraries(shadow-worker PRIVATE Threads::Threads)
add_executable(shadow-candidate src/worker.cpp src/io.cpp)
target_include_directories(shadow-candidate PRIVATE include)
target_compile_definitions(shadow-candidate PRIVATE SHADOW_GOOD_ALPHA=0.65)
target_link_libraries(shadow-candidate PRIVATE Threads::Threads)
string(TOLOWER "${CMAKE_SYSTEM_NAME}" SHADOW_TARGET_OS)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" SHADOW_TARGET_ARCH)
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
  list(LENGTH CMAKE_OSX_ARCHITECTURES SHADOW_ARCH_COUNT)
  if(NOT SHADOW_ARCH_COUNT EQUAL 1)
    message(FATAL_ERROR "Build one target architecture per candidate manifest")
  endif()
  string(TOLOWER "${CMAKE_OSX_ARCHITECTURES}" SHADOW_TARGET_ARCH)
endif()
if(SHADOW_TARGET_ARCH STREQUAL "arm64")
  set(SHADOW_TARGET_ARCH "aarch64")
elseif(SHADOW_TARGET_ARCH STREQUAL "amd64")
  set(SHADOW_TARGET_ARCH "x86_64")
endif()
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:shadow-candidate>/candidate.manifest" CONTENT
"format=1\nartifact=position-filter-v2\nexecutable=./$<TARGET_FILE_NAME:shadow-candidate>\nprotocol=2\narchitecture=${SHADOW_TARGET_ARCH}\nos=${SHADOW_TARGET_OS}\ncontract=position-m-velocity-mps-v1\nreplay_samples=16\nmode=good\nfault_after=0\n")
add_library(shadow_control src/control.cpp)
target_link_libraries(shadow_control PUBLIC shadow_transport)
if(CMAKE_SYSTEM_NAME STREQUAL "QNX")
  target_link_libraries(shadow_control PUBLIC socket)
endif()
add_executable(shadow-demo src/main.cpp src/live_range.cpp)
target_link_libraries(shadow-demo PRIVATE shadow_staging shadow_control)
add_executable(shadowctl src/shadowctl.cpp)
target_link_libraries(shadowctl PRIVATE shadow_control)
include(CTest)
if(BUILD_TESTING)
  add_executable(policy-tests tests/policy_tests.cpp)
  target_link_libraries(policy-tests PRIVATE shadow_core)
  add_test(NAME policy COMMAND policy-tests)
  add_executable(protocol-tests tests/protocol_tests.cpp)
  target_link_libraries(protocol-tests PRIVATE shadow_transport)
  add_test(NAME protocol COMMAND protocol-tests)
  add_executable(staging-tests tests/staging_tests.cpp)
  target_link_libraries(staging-tests PRIVATE shadow_staging)
  add_test(NAME staging COMMAND staging-tests)
  set_tests_properties(staging PROPERTIES TIMEOUT 10)
  if(CMAKE_SYSTEM_NAME STREQUAL "QNX")
    add_executable(qnx-reply-conformance tests/qnx_reply_conformance.cpp)
    target_link_libraries(qnx-reply-conformance PRIVATE shadow_transport)
    add_test(NAME qnx-reply-conformance COMMAND qnx-reply-conformance)
    set_tests_properties(qnx-reply-conformance PROPERTIES TIMEOUT 15)
    add_executable(qnx-cancel-replay-worker tests/qnx_cancel_replay_worker.cpp)
    target_link_libraries(qnx-cancel-replay-worker PRIVATE shadow_transport)
    add_executable(qnx-ipc-receiver tests/qnx_ipc_receiver.cpp)
    target_link_libraries(qnx-ipc-receiver PRIVATE shadow_transport)
    add_executable(qnx-ipc-tests tests/qnx_ipc_tests.cpp)
    target_link_libraries(qnx-ipc-tests PRIVATE shadow_transport)
    foreach(budget_ns IN ITEMS 7500000 112500000)
      add_test(NAME qnx-ipc-fixture-${budget_ns} COMMAND qnx-ipc-tests
        $<TARGET_FILE:qnx-ipc-receiver> ${budget_ns})
      set_tests_properties(qnx-ipc-fixture-${budget_ns} PROPERTIES TIMEOUT 190 SKIP_RETURN_CODE 77)
    endforeach()
  endif()
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    add_test(NAME process-integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:shadow-demo>
      $<TARGET_FILE:shadow-worker>)
    set_tests_properties(process-integration PROPERTIES TIMEOUT 60)
    add_test(NAME live-range-input COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/live_range.py $<TARGET_FILE_DIR:shadow-demo>)
    set_tests_properties(live-range-input PROPERTIES TIMEOUT 20)
    add_test(NAME output-delivery COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/output_delivery.py $<TARGET_FILE:shadow-demo>
      $<TARGET_FILE:shadow-worker>)
    set_tests_properties(output-delivery PROPERTIES TIMEOUT 30)
    add_test(NAME live-deployment COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/deployment.py $<TARGET_FILE_DIR:shadow-demo>)
    # This suite runs several paced broker sessions with their own watchdogs.
    set_tests_properties(live-deployment PROPERTIES TIMEOUT 150)
    add_test(NAME deployment-hook COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/deployment_hook.py $<TARGET_FILE_DIR:shadow-demo>)
    set_tests_properties(deployment-hook PROPERTIES TIMEOUT 45)
    add_test(NAME workbench-api COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/server_tests.py $<TARGET_FILE_DIR:shadow-demo>)
    set_tests_properties(workbench-api PROPERTIES TIMEOUT 30)
    add_test(NAME preboard-pipeline COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/preboard_pipeline.py $<TARGET_FILE_DIR:shadow-demo>)
    set_tests_properties(preboard-pipeline PROPERTIES TIMEOUT 20)
    add_test(NAME qnx-cancel-ordering-verifier COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/qnx_cancel_ordering_verifier.py)
    set_tests_properties(qnx-cancel-ordering-verifier PROPERTIES TIMEOUT 10)
    find_program(SHADOW_NODE_EXECUTABLE node)
    if(SHADOW_NODE_EXECUTABLE)
      add_test(NAME workbench-continuity COMMAND ${SHADOW_NODE_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/workbench_continuity.js)
    endif()
  endif()
endif()
install(TARGETS shadow-demo shadow-worker shadow-candidate shadowctl RUNTIME DESTINATION bin)
install(FILES "$<TARGET_FILE_DIR:shadow-candidate>/candidate.manifest" DESTINATION bin)
install(PROGRAMS tools/deployment_hook_example.sh DESTINATION share/shadowdeploy)
