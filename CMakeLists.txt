cmake_minimum_required(VERSION 3.20)
project(qnx_shadow_deploy VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SHADOW_SANITIZERS "Enable address and undefined behavior sanitizers on the host" OFF)
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(SHADOW_SANITIZERS AND NOT CMAKE_SYSTEM_NAME STREQUAL "QNX")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "QNX")
  add_compile_definitions(_QNX_SOURCE)
endif()
add_library(shadow_core src/policy.cpp)
target_include_directories(shadow_core PUBLIC include)
add_library(shadow_transport src/process.cpp src/io.cpp)
target_include_directories(shadow_transport PUBLIC include)
target_link_libraries(shadow_transport PUBLIC Threads::Threads)
add_executable(shadow-worker src/worker.cpp src/io.cpp)
target_include_directories(shadow-worker PRIVATE include)
target_link_libraries(shadow-worker PRIVATE Threads::Threads)
add_executable(shadow-demo src/main.cpp)
target_link_libraries(shadow-demo PRIVATE shadow_core shadow_transport)
include(CTest)
if(BUILD_TESTING)
  add_executable(policy-tests tests/policy_tests.cpp)
  target_link_libraries(policy-tests PRIVATE shadow_core)
  add_test(NAME policy COMMAND policy-tests)
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    add_test(NAME process-integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:shadow-demo>
      $<TARGET_FILE:shadow-worker>)
    set_tests_properties(process-integration PROPERTIES TIMEOUT 60)
    add_test(NAME workbench-api COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/server_tests.py $<TARGET_FILE_DIR:shadow-demo>)
    set_tests_properties(workbench-api PROPERTIES TIMEOUT 30)
  endif()
endif()
install(TARGETS shadow-demo shadow-worker RUNTIME DESTINATION bin)
